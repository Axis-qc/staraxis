# game 模块文件结构

> 本文档提供 `game/src/main/java/staraxis/game/` 的逐文件列举与职责索引。

```
game/src/main/java/staraxis/game/
|
|-- GameRuntime.java                # 运行时接口：定义游戏模拟主循环的核心约束
|-- SimpleGameRuntime.java          # 简单运行时：基础的同步游戏循环实现
|-- StarAxisGameRuntime.java        # 核心运行时：权威的游戏引擎驱动，负责 Tick 推进、
|                                   #   状态同步与命令执行
|-- 文件结构.md                      # 本文件：提供 game 模块的逐文件列举与职责索引
|
|-- astro/                          # 天体系统：负责恒星系生成与天体管理
|   |-- AstroData.java              # 天体数据集合：持有当前世界中所有恒星系的权威数据
|   |-- AstroGenerator.java         # 天体生成器：根据种子和星区位置生成恒星系、恒星与行星
|   |-- PlanetBody.java             # 行星实体：定义行星的物理属性、轨道数据与地表引用
|   |-- StarBody.java               # 恒星实体：定义恒星的光谱类型、质量与物理参数
|   |-- StarSystem.java             # 恒星系容器：管理系统内的重心、星体与所属关系
|   |-- def/                        # 天体定义/静态配置
|       |-- AstroAssetRepository.java   # 天体资产库：加载并管理天体预设与贴图资源索引
|       |-- OrbitPresetDef.java         # 轨道预设定义：描述行星轨道的分布参数
|       |-- OrbitPresetsFile.java       # 轨道预设存储结构：用于反序列化轨道配置文件
|       |-- OrbitZoneWeightDef.java     # 轨道分区权重定义：按轨道比例分区的行星类型权重表
|       |-- PlanetTypeDef.java          # 行星类型定义：描述不同类别行星的属性范围
|       |-- PresetStarSystemDef.java    # 预设恒星系定义：手动设计的固定恒星系
|       |-- PresetStarSystemRepository.java # 预设恒星系仓库：加载预设系并参与生成
|       |-- PresetStarSystemsFile.java  # 预设恒星系存储结构
|       |-- StarTypeDef.java            # 恒星类型定义：描述不同光谱等级恒星的特征
|
|-- command/                        # 指令系统：处理来自外部的权威修改请求
|   |-- ColonizePlanetCommand.java  # 殖民行星指令
|   |-- ColonizePlanetHandler.java  # 殖民指令处理器
|   |-- Command.java                # 基础指令接口
|   |-- CommandBus.java             # 命令总线：管理指令的注册、排队与顺序执行
|   |-- CommandHandler.java         # 指令处理器接口
|   |-- MoveShipCommand.java        # 舰船移动指令
|   |-- MoveShipHandler.java        # 移动指令处理器
|   |-- SetPlayerTimeStepCommand.java # 玩家步进设置指令
|   |-- SetPlayerTimeStepHandler.java # 步进指令处理器实现
|   |-- SetSystemTimeScaleCommand.java # 系统时间倍率指令
|   |-- SetSystemTimeScaleHandler.java # 倍率指令处理器实现
|
|-- config/                         # 配置系统
|   |-- GlobalConfigRegistry.java   # 全局配置注册表：加载管理各模块配置
|
|-- entity/                         # 实体底层：游戏世界中所有活跃对象的基础架构
|   |-- Entity.java                 # 实体基类：ID、类型、星区坐标、世界位置等通用属性
|   |-- EntityType.java             # 实体类型枚举：定义天体、舰船、重心等所有实体类别
|
|-- government/                     # 政府/体制系统
|   |-- GovernmentDef.java          # 政府定义：描述不同政体的加成与规则
|
|-- intel/                          # 情报系统
|   |-- IntelSystem.java            # 情报系统核心
|   |-- def/
|       |-- IntelConfigDef.java     # 情报配置定义
|       |-- IntelConfigRepository.java # 情报配置仓库
|
|-- log/                            # 日志与性能监控
|   |-- GameLog.java                # 游戏通用日志工具
|   |-- PerformanceMonitor.java     # 性能监控（Tick/渲染耗时采样）
|   |-- TickProfiler.java           # Tick 逐阶段耗时分析器
|
|-- mod/                            # Mod 系统
|   |-- ModManager.java             # Mod 核心管理：处理 mod 注入与加载顺序
|   |-- ModOrder.java               # 加载配置模型
|   |-- ModOrderRepository.java     # 加载配置仓库
|
|-- nation/                         # 国家/势力系统
|   |-- AssetManager.java           # 资产管理器：统一管理 Player 和 Nation 所有权
|   |-- NationDef.java              # 国家定义：储存国家的文化、特性、初始设定等静态数据
|   |-- NationManager.java          # 国家管理器：维护当前游戏中的所有国家实例
|   |-- NationSpawnService.java     # 国家出生点服务：为势力分配初始领土与舰队
|   |-- NationState.java            # 国家运行时状态：人口、科技进度、领地等动态数据
|   |-- SpawnStrategy.java          # 出生策略枚举
|   |-- VisibilitySystem.java       # 视野系统：计算并缓存各国的战争迷雾与星区可见性
|   |-- design/                     # 国家设计工具
|       |-- NationDesignRepository.java # 设计稿仓库
|       |-- PlayerNationDesign.java     # 玩家设计模型
|
|-- planet/                         # 行星经营/地表系统
|   |-- PlanetSurface.java          # 地表模型：管理地表区域、资源站点与城市分布
|   |-- city/                       # 城市系统
|   |   |-- City.java               # 城市实体：人口、建筑、产能与专业化
|   |-- def/                        # 行星/经营静态定义
|   |   |-- CitySpecializationDef.java # 城市专业化定义
|   |   |-- CityStageDef.java       # 城市发展阶段定义
|   |   |-- NamePoolDef.java        # 名称库定义
|   |   |-- PlanetAssetRepository.java # 行星资产仓库
|   |   |-- ResourceTypeDef.java    # 资源类型定义
|   |   |-- SurfaceRegionTypeDef.java # 地表区域定义
|   |-- production/                 # 产能计算逻辑
|   |   |-- CityProductionCalculator.java # 城市产能推导
|   |   |-- ProductionAllocation.java    # 产能分配模型
|   |-- resource/                   # 资源系统
|   |   |-- ResourceSite.java       # 资源矿点/站点
|   |-- surface/                    # 地表区域细节
|       |-- SubSurfaceFeature.java  # 地下特征
|       |-- SurfaceGenerator.java   # 地表生成器：执行行星地形、区域划分的算法
|       |-- SurfaceNamingUtils.java # 地表命名工具
|       |-- SurfaceRegion.java      # 地表区域模型
|
|-- player/                         # 玩家系统
|   |-- PlayerManager.java          # 玩家管理器：注册/查询/状态维护
|   |-- PlayerState.java            # 玩家运行时状态（playerId/displayName/nationId）
|
|-- save/                           # 存档系统
|   |-- WorldSaveService.java       # 世界存档服务
|
|-- server/                         # 头戴式服务端（纯命令行独立运行）
|   |-- HeadlessGameServer.java     # 主入口 main()，init → tickLoop → console
|   |-- ServerConfig.java           # 命令行参数解析（--seed/--stars/--tps）
|   |-- TickLoop.java               # 独立 Tick 循环（固定 TPS，不绑帧率）
|   |-- Console.java                # 开发者控制台（info/entity/system/planets/perf）
|
|-- ship/                           # 舰船/航天系统
|   |-- AbstractShipMovementSystem.java # 抽象移动系统：直飞/滑行通用逻辑
|   |-- FTLTravelSystem.java        # FTL 跃迁系统：星系间超光速移动
|   |-- MovementCommand.java        # 移动指令：序列化客户端下达的移动意图
|   |-- ShipBody.java               # 舰船实体：耐久、能量与组件状态
|   |-- ShipComponent.java          # 舰船组件：模块化设备及功能
|   |-- ShipDesign.java             # 舰船设计图：各模块的布局与规格
|   |-- ShipFullMovementSystem.java # 全量 3D 直飞移动系统
|   |-- ShipMovementSystem.java     # 移动系统主路由（直飞/滑行二模式）
|   |-- ShipSpawnService.java       # 舰船生成服务
|   |-- ShipStatsCalculator.java    # 舰船属性计算器：从组件推导移动/战斗属性
|   |-- def/                        # 舰船定义子包
|   |   |-- ShipAssetRepository.java # 舰船资产仓库
|   |   |-- ShipModuleDef.java      # 舰船模块定义
|   |   |-- ShipModulesFile.java    # 舰船模块配置文件
|   |-- design/                     # 舰船设计子包
|       |-- ShipDesignRepository.java # 舰船设计图仓库
|
|-- sim/                            # 模拟驱动/时钟系统
|   |-- LptAssigner.java            # LPT 任务分配器
|   |-- SimulationClock.java        # 模拟时钟工具：Tick 转换、时间步长计算与日期切分
|   |-- SimulationTime.java         # 时间数据结构：simulationTick 与游戏日期
|   |-- SystemLoad.java             # 系统负载模型
|   |-- TickDispatcher.java         # Tick 分发器：多线程任务调度
|   |-- TimelineSystem.java         # 时间线系统
|   |-- WorkerLocalBuffer.java      # 工作线程本地缓冲区
|
|-- space/                          # 太空/空间系统
|   |-- OrbitalElements.java        # 开普勒轨道根数
|   |-- OrbitSolver.java            # 轨道求解器：由轨道根数+时间求位置
|   |-- SpacePosition.java          # 3D 空间位置
|   |-- event/                      # 太空事件
|   |   |-- CrossSystemEvent.java       # 跨星系事件
|   |   |-- CrossSystemEventTable.java  # 跨星系事件表
|   |-- galaxy/                     # 星系生成
|   |   |-- EllipticalGalaxyGenerator.java # 椭圆星系生成器
|   |   |-- GalaxyConfig.java       # 星系配置
|   |   |-- GalaxyData.java         # 星系数据
|   |   |-- GalaxyGenerator.java    # 星系生成器接口
|   |   |-- GalaxyGeneratorFactory.java # 工厂
|   |   |-- GalaxyType.java         # 星系类型枚举
|   |   |-- IrregularGalaxyGenerator.java # 不规则星系生成器
|   |   |-- SpectralType.java       # 光谱类型
|   |   |-- SpiralGalaxyGenerator.java  # 旋涡星系生成器（默认）
|   |   |-- StarPosition.java       # 恒星位置
|   |-- octree/                     # 空间分区
|       |-- GalaxyOctree.java       # 星系八叉树
|       |-- OctreeNode.java         # 八叉树节点
|
|-- species/                        # 种族系统
|   |-- SpeciesDef.java             # 种族定义：生理特性、环境偏好等
|
|-- state/                          # 状态管理/快照系统
|   |-- DailySettlementState.java   # 日结算状态
|   |-- DailySettlementStateBuffer.java # 双缓冲容器（跨线程安全发布）
|   |-- RealTimeWorldState.java     # 实时世界状态：每帧同步的实体快照
|   |-- RealTimeWorldStateBuffer.java # 双缓冲容器（高频实时快照）
|   |-- WorldState.java             # 权威世界容器：模拟层的唯一真相源
|   |-- snapshot/                   # 快照详情模型
|       |-- EntitySnapshot.java     # 实体快照基类
|       |-- ShipDetails.java        # 舰船快照详情
|       |-- StarSnapshot.java       # 恒星快照详情
|
|-- tech/                           # 科技系统
|   |-- TechDef.java                # 科技项定义
|
|-- util/                           # 通用工具
|   |-- ProgressCallback.java       # 进度回调接口
|   |-- WeightedRandomUtil.java     # 加权随机抽取工具
|
|-- world/                          # 世界/地理基础
    |-- WorldConstants.java         # 世界常量（AU_IN_GU、SECTOR_SIZE_GU 等核心参数）
    |-- WorldGenConfig.java         # 世界生成配置
    |-- WorldType.java              # 世界类型枚举
```
